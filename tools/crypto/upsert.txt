BEGIN;

-- Create table if it doesn't exist (safe no-op if already created)
CREATE TABLE IF NOT EXISTS public.credentials (
  scope_type   text    NOT NULL,     -- e.g., 'account' or 'broker'
  scope_id     text    NOT NULL,     -- e.g., 'alpaca:paper'
  key          text    NOT NULL,     -- e.g., 'api_key' | 'api_secret' | 'base_url'
  value_cipher bytea   NOT NULL,     -- Fernet ciphertext as bytea (from base64)
  created_at   timestamptz NOT NULL DEFAULT now(),
  updated_at   timestamptz NOT NULL DEFAULT now(),
  CONSTRAINT pk_credentials PRIMARY KEY(scope_type, scope_id, key)
);

-- Keep updated_at fresh on change
CREATE OR REPLACE FUNCTION public.t_credentials_touch_updated_at()
RETURNS trigger LANGUAGE plpgsql AS $$
BEGIN
  NEW.updated_at := now();
  RETURN NEW;
END
$$;

DROP TRIGGER IF EXISTS trg_credentials_touch ON public.credentials;
CREATE TRIGGER trg_credentials_touch
BEFORE UPDATE ON public.credentials
FOR EACH ROW EXECUTE FUNCTION public.t_credentials_touch_updated_at();
